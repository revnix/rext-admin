### Date Formatters

```typescript
import { dateFormat } from '@/lib/formatters/date-formatters';

dateFormat.short(date)           // "Jan 15, 2024"
dateFormat.long(date)            // "January 15, 2024"
dateFormat.shortWithTime(date)   // "Jan 15, 2024 at 3:45 PM"
dateFormat.longWithTime(date)    // "January 15, 2024 at 3:45 PM"
dateFormat.relative(date)        // "2 hours ago"
dateFormat.timeOnly(date)        // "3:45 PM"
dateFormat.numeric(date)         // "01/15/2024"
dateFormat.iso(date)             // "2024-01-15"
dateFormat.custom(date, format)  // Custom format
```

**Benefits:**
- Null-safe (returns "" for null)
- Consistent across app
- Uses date-fns internally
- TypeScript support

### Number Formatters

## Migration Pattern

### Before (typical component):
```typescript
import { format, formatDistanceToNow } from "date-fns";
import { toast } from "sonner";

function Component({ item }) {
  const removeItem = useStore((state) => state.removeItem);

  const handleDelete = async () => {
    try {
      await service.delete(item.workspace_id, item.id);
      removeItem(item.id);
      toast.success("Item deleted successfully");
    } catch (error) {
      log.error("Failed to delete:", error);
      toast.error("Failed to delete item");
    }
  };

  const formattedDate = new Date(item.created_at).toLocaleDateString("en-US", {
    month: "short",
    day: "numeric",
    year: "numeric",
  });

  const wordCount = item.word_count >= 1000
    ? `${(item.word_count / 1000).toFixed(1)}k`
    : item.word_count.toString();

  return (
    <div>
      <p>{formattedDate}</p>
      <p>{wordCount} words</p>
      <Button onClick={handleDelete}>Delete</Button>
    </div>
  );
}
```

### After (refactored):
```typescript
import { useDeleteHandler } from "@/hooks/useDeleteHandler";
import { dateFormat, numberFormat } from "@/lib/formatters";

function Component({ item }) {
  const removeItem = useStore((state) => state.removeItem);

  const { handleDelete, isDeleting } = useDeleteHandler({
    deleteFunction: (id) => service.delete(item.workspace_id, id),
    resourceName: "item",
    onSuccess: () => removeItem(item.id),
  });

  return (
    <div>
      <p>{dateFormat.short(item.created_at)}</p>
      <p>{numberFormat.compact(item.word_count)} words</p>
      <Button onClick={() => handleDelete(item.id)} disabled={isDeleting}>
        {isDeleting ? "Deleting..." : "Delete"}
      </Button>
    </div>
  );
}
```

**Improvements:**
- 30% fewer lines
- More readable
- Consistent formatting
- Better error handling
- Centralized logic
