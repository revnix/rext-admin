Visual comparison of code improvements from Phase 1.3 & 1.4 refactoring.

## Example 1: Web Knowledge Card - Delete Logic

### Before (BaseKnowledgeCard integration)

```typescript
// web-knowledge-card.tsx - OLD
const removeItem = useWebKnowledgeStore((state) => state.removeItem);

const cardConfig: KnowledgeCardConfig<WebKnowledge> = {
  // ... other config
  onDelete: async (item) => {
    await webKnowledgeService.delete(item.workspace_id, item.id);
    removeItem(item.id);
    toast.success("Web knowledge deleted successfully");
  },
};
```

**Lines**: 8 lines of delete logic

**Issues**:
- No loading state
- No error handling
- Manual toast management
- Duplicated across card + list variants

---

### After (with useDeleteHandler)

```typescript
// web-knowledge-card.tsx - NEW
const removeItem = useWebKnowledgeStore((state) => state.removeItem);

const { handleDelete: deleteWebKnowledge } = useDeleteHandler({
  deleteFunction: (id) => webKnowledgeService.delete(item.workspace_id, id),
  resourceName: "web knowledge",
  onSuccess: () => {
    removeItem(item.id);
  },
});

const cardConfig: KnowledgeCardConfig<WebKnowledge> = {
  // ... other config
  onDelete: async (item) => {
    await deleteWebKnowledge(item.id);
  },
};
```

**Lines**: 11 lines total (clearer separation of concerns)

**Improvements**:
- ✅ Automatic loading state (`isDeleting`)
- ✅ Built-in error handling with logging
- ✅ Automatic toast notifications
- ✅ Type-safe
- ✅ Consistent across variants

**Net effect**: Same lines but much better UX and maintainability

## Example 2: Text Knowledge Card - Complete Refactor

### Before

```typescript
// text-knowledge-card.tsx - OLD
import { toast } from "sonner";
import {
  formatCountLocale,
  formatDateCustom,
} from "./shared";

// Helper function
const formatDate = (dateString: string) => {
  return formatDateCustom(dateString, {
    year: "numeric",
    month: "short",
    day: "numeric",
  });
};

export function TextKnowledgeCard({ item, onEdit }) {
  const removeItem = useTextKnowledgeStore((state) => state.removeItem);

  const cardConfig: KnowledgeCardConfig<TextKnowledge> = {
    getDescription: (item) => (
      <div className="flex items-center gap-1">
        <Calendar className="h-3 w-3" />
        {formatDate(item.created_at)}
        {item.updated_at && item.updated_at !== item.created_at && (
          <span className="text-xs">• Updated {formatDate(item.updated_at)}</span>
        )}
      </div>
    ),
    getMetadataSections: (item) => [
      {
        id: "statistics",
        condition: !!(item.word_count || item.char_count),
        content: (
          <div className="text-xs text-muted-foreground space-y-1">
            {item.word_count && <div>{formatCountLocale(item.word_count)} words</div>}
            {item.char_count && <div>{formatCountLocale(item.char_count)} characters</div>}
          </div>
        ),
      },
    ],
    onDelete: async (item) => {
      await textKnowledgeService.delete(item.workspace_id, item.id);
      removeItem(item.id);
      toast.success(`Text note "${item.title}" deleted successfully`);
    },
  };

  return <BaseKnowledgeCard item={item} config={cardConfig} />;
}
```

**Lines**: ~48 lines
**Issues**:
- Custom helper function needed
- Multiple formatter imports
- Manual delete handling
- No error handling

---

### After

```typescript
        {item.updated_at && item.updated_at !== item.created_at && (
          <span className="text-xs">• Updated {dateFormat.short(item.updated_at)}</span>
        )}
      </div>
    ),
    getMetadataSections: (item) => [
      {
        id: "statistics",
        condition: !!(item.word_count || item.char_count),
        content: (
          <div className="text-xs text-muted-foreground space-y-1">
            {item.word_count && <div>{numberFormat.integer(item.word_count)} words</div>}
            {item.char_count && <div>{numberFormat.integer(item.char_count)} characters</div>}
          </div>
        ),
      },
    ],
    onDelete: async (item) => {
      await deleteTextKnowledge(item.id);
    },
  };

  return <BaseKnowledgeCard item={item} config={cardConfig} />;
}
```

**Lines**: ~43 lines
**Improvements**:
- ❌ No helper function needed (removed 7 lines)
- ✅ Cleaner imports (centralized formatters)
- ✅ Better delete handling with error/loading
- ✅ More descriptive formatter names
- ✅ Consistent with other components

**Lines saved**: ~5 lines + removed helper function

## Example 4: Complete Component Comparison

### Before - Full Knowledge Card Component

```typescript
"use client";

import { Calendar, Edit2, FileText, Tag, Trash2 } from "lucide-react";
import { toast } from "sonner";
import { Badge } from "@/components/ui/badge";
import { textKnowledgeService } from "@/services/knowledge-api";
import { useTextKnowledgeStore } from "@/stores/knowledge-store";
import type { TextKnowledge } from "@/types/workspace";
import {
  BaseKnowledgeCard,
  type KnowledgeCardConfig,
  truncateContent,
  formatCountLocale,
  formatDateCustom,
} from "./shared";

// Helper function to format date
const formatDate = (dateString: string) => {
  return formatDateCustom(dateString, {
    year: "numeric",
    month: "short",
    day: "numeric",
  });
};

export function TextKnowledgeCard({ item, onSelect, onEdit, isSelected = false }) {
  const removeItem = useTextKnowledgeStore((state) => state.removeItem);

  const cardConfig: KnowledgeCardConfig<TextKnowledge> = {
    primaryIcon: FileText,
    getTitle: (item) => item.title,
    getDescription: (item) => (
      <div className="flex items-center gap-1">
        <Calendar className="h-3 w-3" />
        {formatDate(item.created_at)}
        {item.updated_at && item.updated_at !== item.created_at && (
          <span className="text-xs">• Updated {formatDate(item.updated_at)}</span>
        )}
      </div>
    ),
    getStatusConfig: () => null,
    getActions: (item) => [
      {
        icon: Edit2,
        label: "Edit",
        onClick: (e) => {
          e.stopPropagation();
          onEdit?.(item);
        },
      },
      {
        icon: Trash2,
        label: "Delete",
        variant: "destructive" as const,
        onClick: () => {},
        confirmationConfig: {
          title: "Delete Text Note",
          description: `Are you sure you want to delete "${item.title}"?`,
          confirmText: "Delete",
        },
      },
    ],
    getMetadataSections: (item) => [
      {
        id: "content-preview",
        content: (
          <div className="text-sm text-muted-foreground line-clamp-3">
            {truncateContent(item.content)}
          </div>
        ),
      },
      {
        id: "tags",
        condition: !!(item.tags && item.tags.length > 0),
        content: (
          <div className="flex flex-wrap gap-1">
            {item.tags?.slice(0, 3).map((tag) => (
              <Badge key={tag} variant="outline">
                <Tag className="mr-1 h-2 w-2" />
                {tag}
              </Badge>
            ))}
          </div>
        ),
      },
      {
        id: "statistics",
        condition: !!(item.word_count || item.char_count),
        content: (
          <div className="text-xs text-muted-foreground space-y-1">
            {item.word_count && <div>{formatCountLocale(item.word_count)} words</div>}
            {item.char_count && <div>{formatCountLocale(item.char_count)} characters</div>}
          </div>
        ),
      },
    ],
    onDelete: async (item) => {
      await textKnowledgeService.delete(item.workspace_id, item.id);
      removeItem(item.id);
      toast.success(`Text note "${item.title}" deleted successfully`);
    },
  };

  return (
    <BaseKnowledgeCard
      item={item}
      config={cardConfig}
      onSelect={onSelect}
      isSelected={isSelected}
    />
  );
}
```

**Total**: ~105 lines

## Formatter Usage Comparison

### Number Formatting

```typescript
// BEFORE - Various patterns across codebase
item.word_count.toLocaleString()
item.word_count >= 1000 ? `${(item.word_count / 1000).toFixed(1)}k` : item.word_count.toString()
count ? count.toLocaleString() : "0"

// AFTER - Consistent, descriptive
numberFormat.integer(item.word_count)    // "1,234"
numberFormat.compact(item.word_count)    // "1.2K"
numberFormat.decimal(price, 2)           // "1,234.56"
numberFormat.currency(price)             // "$1,234.56"
numberFormat.fileSize(bytes)             // "1.2 MB"
```

### Date Formatting

```typescript
// BEFORE - Verbose, inconsistent
new Date(date).toLocaleDateString("en-US", {
  month: "short",
  day: "numeric",
  year: "numeric",
})
format(new Date(date), "MMM d, yyyy")
formatDistanceToNow(date, { addSuffix: true })

// AFTER - Clean, self-documenting
dateFormat.short(date)           // "Jan 15, 2024"
dateFormat.shortWithTime(date)   // "Jan 15, 2024 at 3:45 PM"
dateFormat.relative(date)        // "2 hours ago"
```
