```typescript
// /lib/topic-builder-utils.ts
import { TopicBuilderFormData } from '../types/topic-builder';
import { industryAudienceMap, ymylIndustries } from '../data/topic-builder-options';

export function detectYMYL(industry: string): boolean {
  return ymylIndustries.includes(industry);
}

export function getAudienceForIndustry(industry: string): string[] {
  return industryAudienceMap[industry] || [];
}

export function validateFormStep(step: number, data: Partial<TopicBuilderFormData>): boolean {
  switch(step) {
    case 1: // Wizard Mode Selection
      return !!data.wizardMode;
    case 2: // Industry/Subject
      if (data.wizardMode === 'subject-first') {
        return !!data.subject && !!data.industry;
      }
      return !!data.industry;
    // Additional step validations...
  }
}

// /data/topic-builder-options.ts
export const industries = [
  { value: 'technology', label: 'Technology' },
  { value: 'healthcare', label: 'Healthcare' },
  // More industries...
];

export const ymylIndustries = ['healthcare', 'finance', 'legal'];

export const industryAudienceMap = {
  'technology': ['developers', 'IT professionals', 'tech enthusiasts'],
  'healthcare': ['patients', 'healthcare providers', 'medical professionals'],
  // More mappings...
};
```
