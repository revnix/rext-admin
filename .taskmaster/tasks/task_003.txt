```typescript
// /hooks/use-topic-builder.ts
import { useState } from 'react';
import { TopicBuilderFormData } from '../types/topic-builder';
import { validateFormStep } from '../lib/topic-builder-utils';

export function useTopicBuilder() {
  const [currentStep, setCurrentStep] = useState(1);
  const [formData, setFormData] = useState<Partial<TopicBuilderFormData>>({
    wizardMode: 'subject-first',
    num_ideas: 5,
  });
  const [errors, setErrors] = useState<Record<string, string>>({});

  const updateFormData = (newData: Partial<TopicBuilderFormData>) => {
    setFormData(prev => ({ ...prev, ...newData }));
  };

  const goToNextStep = () => {
    if (validateFormStep(currentStep, formData)) {
      setCurrentStep(prev => prev + 1);
      setErrors({});
    } else {
      // Set validation errors
    }
  };

  const goToPrevStep = () => {
    setCurrentStep(prev => Math.max(1, prev - 1));
  };

  return {
    currentStep,
    formData,
    errors,
    updateFormData,
    goToNextStep,
    goToPrevStep,
  };
}

// /components/topic-builder/steps/IndustryStep.tsx
import { industries } from '../../../data/topic-builder-options';

export function IndustryStep({ formData, updateFormData }) {
  return (
    <div className="step-container">
      <h2>Select Your Industry</h2>
      <select
        value={formData.industry || ''}
        onChange={(e) => updateFormData({ industry: e.target.value })}
      >
        <option value="">Select an industry...</option>
        {industries.map(industry => (
          <option key={industry.value} value={industry.value}>
            {industry.label}
          </option>
        ))}
      </select>
      
      {formData.wizardMode === 'subject-first' && (
        <div>
          <h3>What's your specific subject?</h3>
          <input
            type="text"
            value={formData.subject || ''}
            onChange={(e) => updateFormData({ subject: e.target.value })}
            placeholder="Enter your subject or keyword"
          />
        </div>
      )}
    </div>
  );
}
```
