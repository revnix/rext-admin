// /lib/ai-service.ts
import OpenAI from 'openai';
import { TopicBuilderFormData } from '../types/topic-builder';

const openai = new OpenAI({
  apiKey: process.env.OPENAI_API_KEY,
});

export async function generateTopicsWithAI(formData: TopicBuilderFormData) {
  try {
    const prompt = buildPromptFromFormData(formData);
    
    const response = await openai.chat.completions.create({
      model: 'gpt-4-turbo',
      messages: [
        { role: 'system', content: 'You are a specialized content topic generator.' },
        { role: 'user', content: prompt }
      ],
      response_format: { type: 'json_object' },
      temperature: 0.7,
    });
    
    const result = JSON.parse(response.choices[0].message.content);
    return {
      topics: result.topics,
      request_id: response.id,
      model_used: 'gpt-4-turbo',
      generation_time_ms: response.usage.completion_tokens * 10, // Approximation
    };
  } catch (error) {
    console.error('OpenAI API error:', error);
    // Try fallback to Groq if OpenAI fails
    return generateTopicsWithGroq(formData);
  }
}

function buildPromptFromFormData(formData: TopicBuilderFormData): string {
  const { wizardMode, industry, subject, content_type, audience, purpose, tone } = formData;
  
  let prompt = '';
  
  if (wizardMode === 'subject-first') {
    prompt = `Generate ${formData.num_ideas} content topics about ${subject} for the ${industry} industry.`;
  } else {
    prompt = `Generate ${formData.num_ideas} content topics for the ${industry} industry.`;
  }
  
  prompt += `\n\nContent type: ${content_type}`;
  if (formData.platform) prompt += `\nPlatform: ${formData.platform}`;
  prompt += `\nTarget audience: ${audience.join(', ')}`;
  prompt += `\nContent purpose: ${purpose.join(', ')}`;
  prompt += `\nTone: ${tone.join(', ')}`;
  
  if (formData.keywords) prompt += `\nInclude these keywords: ${formData.keywords}`;
  if (formData.exclude) prompt += `\nAvoid these topics: ${formData.exclude}`;
  
  prompt += `\n\nReturn the results as a JSON object with this structure:
{
  "topics": [
    {
      "id": "unique-id",
      "title": "Topic title",
      "angle": "Brief description of the angle",
      "channel_fit": ["platforms this works well for"],
      "audience_fit": ["audiences this appeals to"],
      "why_it_works": "Explanation of why this topic works",
      "scores": {
        "relevance": 0.95,
        "freshness": 0.85,
        "novelty": 0.75
      },
      "tags": ["relevant", "tags"]
    }
  ]
}`;
  
  return prompt;
}

// /app/api/generate-topics/route.ts
import { NextResponse } from 'next/server';
import { generateTopicsWithAI } from '@/lib/ai-service';
import { TopicBuilderFormData } from '@/types/topic-builder';

export async function POST(request: Request) {
  try {
    const body = await request.json();
    const formData = body.formData as TopicBuilderFormData;
    
    // Validate the request data
    if (!formData || !formData.industry) {
      return NextResponse.json(
        { error: 'Invalid request data' },
        { status: 400 }
      );
    }
    
    const result = await generateTopicsWithAI(formData);
    
    return NextResponse.json({
      topics: result.topics,
      request_id: result.request_id,
      generated_at: new Date().toISOString(),
      model_used: result.model_used,
      generation_time_ms: result.generation_time_ms
    });
  } catch (error) {
    console.error('Topic generation error:', error);
    return NextResponse.json(
      { 
        error: 'Failed to generate topics',
        error_code: 'generation_failed',
        fallback_available: true
      },
      { status: 500 }
    );
  }
}
