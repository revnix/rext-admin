```typescript
// /components/topic-builder/results/TopicFilters.tsx
import { useState } from 'react';

interface TopicFiltersProps {
  sortBy: 'relevance' | 'freshness' | 'novelty';
  onSortChange: (sort: 'relevance' | 'freshness' | 'novelty') => void;
  availableTags: string[];
  selectedTags: string[];
  onTagsChange: (tags: string[]) => void;
}

export function TopicFilters({ 
  sortBy, 
  onSortChange, 
  availableTags, 
  selectedTags, 
  onTagsChange 
}: TopicFiltersProps) {
  const [showTagFilter, setShowTagFilter] = useState(false);
  
  const handleTagToggle = (tag: string) => {
    if (selectedTags.includes(tag)) {
      onTagsChange(selectedTags.filter(t => t !== tag));
    } else {
      onTagsChange([...selectedTags, tag]);
    }
  };
  
  return (
    <div className="topic-filters">
      <div className="sort-controls">
        <label>Sort by:</label>
        <div className="sort-buttons">
          <button
            className={sortBy === 'relevance' ? 'active' : ''}
            onClick={() => onSortChange('relevance')}
          >
            Relevance
          </button>
          <button
            className={sortBy === 'freshness' ? 'active' : ''}
            onClick={() => onSortChange('freshness')}
          >
            Freshness
          </button>
          <button
            className={sortBy === 'novelty' ? 'active' : ''}
            onClick={() => onSortChange('novelty')}
          >
            Novelty
          </button>
        </div>
      </div>
      
      <div className="tag-filters">
        <button 
          className="filter-toggle"
          onClick={() => setShowTagFilter(!showTagFilter)}
        >
          Filter by Tags {showTagFilter ? '▲' : '▼'}
        </button>
        
        {showTagFilter && (
          <div className="tag-options">
            {availableTags.map(tag => (
              <label key={tag} className="tag-option">
                <input
                  type="checkbox"
                  checked={selectedTags.includes(tag)}
                  onChange={() => handleTagToggle(tag)}
                />
                <span>{tag}</span>
              </label>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}

interface ScoreVisualizationProps {
  relevance: number;
  freshness: number;
  novelty: number;
  size?: 'small' | 'medium' | 'large';
}

export function ScoreVisualization({ 
  relevance, 
  freshness, 
  novelty, 
  size = 'medium' 
}: ScoreVisualizationProps) {
  const getScoreColor = (score: number) => {
    if (score >= 0.8) return 'var(--color-success)';
    if (score >= 0.6) return 'var(--color-warning)';
    return 'var(--color-danger)';
  };
  
  const sizeClass = `score-viz-${size}`;
  
  return (
    <div className={`score-visualization ${sizeClass}`}>
      <div className="score-item">
        <span className="score-label">Relevance</span>
        <div className="score-bar-container">
          <div 
            className="score-bar" 
            style={{ 
              width: `${relevance * 100}%`,
              backgroundColor: getScoreColor(relevance)
            }} 
          />
        </div>
        <span className="score-value">{Math.round(relevance * 100)}%</span>

      <div className="score-item">

// Enhanced TopicList with advanced filtering
import { useState, useMemo } from 'react';
import { TopicCard } from './TopicCard';
import { TopicFilters } from './TopicFilters';
import { GeneratedTopic } from '@/types/topic-builder';

interface TopicListProps {
  topics: GeneratedTopic[];
  onSaveTopic: (id: string) => void;
  onDiscardTopic: (id: string) => void;
}

export function TopicList({ topics, onSaveTopic, onDiscardTopic }: TopicListProps) {
  const [sortBy, setSortBy] = useState<'relevance' | 'freshness' | 'novelty'>('relevance');
  const [selectedTags, setSelectedTags] = useState<string[]>([]);
  const [selectedAudiences, setSelectedAudiences] = useState<string[]>([]);
  const [minScore, setMinScore] = useState(0);
  
  // Get all available tags and audiences from topics
  const availableTags = useMemo(() => {
    return Array.from(new Set(topics.flatMap(topic => topic.tags)));
  }, [topics]);
  
  const availableAudiences = useMemo(() => {
    return Array.from(new Set(topics.flatMap(topic => topic.audience_fit)));
  }, [topics]);
  
  // Filter and sort topics
  const filteredTopics = useMemo(() => {
    return topics
      .filter(topic => {
        // Filter by tags
        if (selectedTags.length > 0 && !selectedTags.some(tag => topic.tags.includes(tag))) {
          return false;
        }
        
        // Filter by audiences
        if (selectedAudiences.length > 0 && !selectedAudiences.some(audience => topic.audience_fit.includes(audience))) {
          return false;
        }
        
        // Filter by minimum score
        if (topic.scores[sortBy] < minScore) {
          return false;
        }
        
        return true;
      })
      .sort((a, b) => b.scores[sortBy] - a.scores[sortBy]);
  }, [topics, selectedTags, selectedAudiences, minScore, sortBy]);
  
  return (
    <div className="topic-list-container">
      <div className="topic-list-header">
        <h2>Generated Topics ({filteredTopics.length} of {topics.length})</h2>
        
        <TopicFilters
          sortBy={sortBy}
          onSortChange={setSortBy}
          availableTags={availableTags}
          selectedTags={selectedTags}
          onTagsChange={setSelectedTags}
        />
        
        <div className="advanced-filters">
          <div className="audience-filter">
            <h3>Filter by Audience</h3>
            <div className="audience-options">
              {availableAudiences.map(audience => (
                <label key={audience} className="audience-option">
                  <input
                    type="checkbox"
                    checked={selectedAudiences.includes(audience)}
                    onChange={() => {
                      if (selectedAudiences.includes(audience)) {
                        setSelectedAudiences(selectedAudiences.filter(a => a !== audience));
                      } else {
                        setSelectedAudiences([...selectedAudiences, audience]);
                      }
                    }}
                  />
                  <span>{audience}</span>
                </label>
              ))}
            </div>
          </div>
          
          <div className="score-filter">
            <h3>Minimum {sortBy.charAt(0).toUpperCase() + sortBy.slice(1)} Score</h3>
            <input
              type="range"
              min="0"
              max="1"
              step="0.1"
              value={minScore}
              onChange={(e) => setMinScore(parseFloat(e.target.value))}
            />
            <span>{Math.round(minScore * 100)}%</span>
          </div>
        </div>
      </div>
      
      <div className="topic-list">
        {filteredTopics.length > 0 ? (
          filteredTopics.map(topic => (
            <TopicCard
              key={topic.id}
              topic={topic}
              onSave={() => onSaveTopic(topic.id)}
              onDiscard={() => onDiscardTopic(topic.id)}
            />
          ))
        ) : (
          <div className="no-topics-message">
            <p>No topics match your current filters.</p>
            <button onClick={() => {
              setSelectedTags([]);
              setSelectedAudiences([]);
              setMinScore(0);
            }}>
              Clear All Filters
            </button>
          </div>
        )}
      </div>
    </div>
  );
}
```
