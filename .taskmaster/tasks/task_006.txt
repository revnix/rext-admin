// Enhanced version of /components/topic-builder/steps/AudienceStep.tsx
import { useEffect } from 'react';
import { getAudienceForIndustry } from '@/lib/topic-builder-utils';
import { TopicBuilderFormData } from '@/types/topic-builder';

interface AudienceStepProps {
  formData: Partial<TopicBuilderFormData>;
  updateFormData: (data: Partial<TopicBuilderFormData>) => void;
}

export function AudienceStep({ formData, updateFormData }: AudienceStepProps) {
  // Get audience options based on selected industry
  const audienceOptions = formData.industry ? 
    getAudienceForIndustry(formData.industry) : [];
  
  // Update audience selection when industry changes
  useEffect(() => {
    if (formData.industry && (!formData.audience || formData.audience.length === 0)) {
      // Set default audience based on industry
      const defaultAudiences = audienceOptions.slice(0, 2); // First two options as default
      updateFormData({ audience: defaultAudiences });
    }
  }, [formData.industry, formData.audience, audienceOptions, updateFormData]);
  
  const handleAudienceChange = (audience: string, checked: boolean) => {
    const currentAudiences = formData.audience || [];
    if (checked) {
      updateFormData({ 
        audience: [...currentAudiences, audience] 
      });
    } else {
      updateFormData({ 
        audience: currentAudiences.filter(a => a !== audience) 
      });
    }
  };
  
  return (
    <div className="step-container">
      <h2>Select Your Target Audience</h2>
      <p>Who will be consuming this content?</p>
      
      <div className="audience-options">
        {audienceOptions.map(audience => (
          <label key={audience} className="audience-option">
            <input
              type="checkbox"
              checked={(formData.audience || []).includes(audience)}
              onChange={(e) => handleAudienceChange(audience, e.target.checked)}
            />
            <span>{audience}</span>
          </label>
        ))}
        
        <label className="audience-option custom">
          <input
            type="checkbox"
            checked={!!(formData.audience_other)}
            onChange={(e) => {
              if (!e.target.checked) {
                updateFormData({ audience_other: undefined });
              }
            }}
          />
          <span>Other:</span>
          <input
            type="text"
            value={formData.audience_other || ''}
            onChange={(e) => {
              const value = e.target.value;
              updateFormData({ 
                audience_other: value,
                audience: value ? 
                  [...(formData.audience || []).filter(a => a !== formData.audience_other), value] :
                  (formData.audience || []).filter(a => a !== formData.audience_other)
              });
            }}
            placeholder="Custom audience"
            disabled={!formData.audience_other && !(formData.audience || []).some(a => a === formData.audience_other)}
          />
        </label>
      </div>
      
      <div className="reader-level-selection">
        <h3>Reader Knowledge Level</h3>
        <div className="toggle-group">
          <button
            className={formData.reader_level === 'beginner' ? 'active' : ''}
            onClick={() => updateFormData({ reader_level: 'beginner' })}
          >
            Beginner
          </button>
          <button
            className={formData.reader_level === 'intermediate' ? 'active' : ''}
            onClick={() => updateFormData({ reader_level: 'intermediate' })}
          >
            Intermediate
          </button>
          <button
            className={formData.reader_level === 'expert' ? 'active' : ''}
            onClick={() => updateFormData({ reader_level: 'expert' })}
          >
            Expert
          </button>
          <button
            className={formData.reader_level === 'mixed' ? 'active' : ''}
            onClick={() => updateFormData({ reader_level: 'mixed' })}
          >
            Mixed
          </button>
        </div>
      </div>
    </div>
  );
}

// Enhanced version of /components/topic-builder/steps/ContentFormatStep.tsx
import { useEffect } from 'react';
import { contentTypes, platformsByContentType } from '@/data/topic-builder-options';
import { TopicBuilderFormData } from '@/types/topic-builder';

interface ContentFormatStepProps {
  formData: Partial<TopicBuilderFormData>;
  updateFormData: (data: Partial<TopicBuilderFormData>) => void;
}

export function ContentFormatStep({ formData, updateFormData }: ContentFormatStepProps) {
  // Get platform options based on selected content type
  const platformOptions = formData.content_type ? 
    platformsByContentType[formData.content_type] : [];
  
  // Update platform when content type changes
  useEffect(() => {
    if (formData.content_type && formData.platform) {
      // Check if current platform is valid for new content type
      const isValidPlatform = platformOptions.some(p => p.value === formData.platform);
      if (!isValidPlatform) {
        // Reset platform if not valid for new content type
        updateFormData({ platform: undefined });
      }
    }
  }, [formData.content_type, formData.platform, platformOptions, updateFormData]);
  
  return (
    <div className="step-container">
      <h2>Content Format & Distribution</h2>
      <p>What type of content are you creating?</p>
      
      <div className="content-type-selection">
        <select
          value={formData.content_type || ''}
          onChange={(e) => updateFormData({ content_type: e.target.value })}
        >
          <option value="">Select content type...</option>
          {contentTypes.map(type => (
            <option key={type.value} value={type.value}>
              {type.label}
            </option>
          ))}
        </select>
      </div>
      
      {formData.content_type && (
        <div className="platform-selection">
          <h3>Distribution Platform</h3>
          <p>Where will this content be published?</p>
          
          <select
            value={formData.platform || ''}
            onChange={(e) => updateFormData({ platform: e.target.value })}
          >
            <option value="">Select platform...</option>
            {platformOptions.map(platform => (
              <option key={platform.value} value={platform.value}>
                {platform.label}
              </option>
            ))}
          </select>
        </div>
      )}
    </div>
  );
}
