// /components/topic-builder/results/TopicList.tsx
import { useState } from 'react';
import { TopicCard } from './TopicCard';
import { TopicFilters } from './TopicFilters';
import { GeneratedTopic } from '@/types/topic-builder';

interface TopicListProps {
  topics: GeneratedTopic[];
  onSaveTopic: (id: string) => void;
  onDiscardTopic: (id: string) => void;
  onExportTopics: () => void;
  onRegenerateTopic: (id: string) => void;
}

export function TopicList({ topics, onSaveTopic, onDiscardTopic, onExportTopics, onRegenerateTopic }: TopicListProps) {
  const [sortBy, setSortBy] = useState<'relevance' | 'freshness' | 'novelty'>('relevance');
  const [filterTags, setFilterTags] = useState<string[]>([]);
  
  // Sort and filter topics
  const filteredTopics = topics
    .filter(topic => {
      if (filterTags.length === 0) return true;
      return filterTags.some(tag => topic.tags.includes(tag));
    })
    .sort((a, b) => b.scores[sortBy] - a.scores[sortBy]);
  
  return (
    <div className="topic-list-container">
      <div className="topic-list-header">
        <h2>Generated Topics</h2>
        <TopicFilters
          sortBy={sortBy}
          onSortChange={setSortBy}
          availableTags={Array.from(new Set(topics.flatMap(t => t.tags)))}
          selectedTags={filterTags}
          onTagsChange={setFilterTags}
        />
        <button onClick={onExportTopics} className="export-button">
          Export Topics
        </button>
      </div>
      
      <div className="topic-list">
        {filteredTopics.map(topic => (
          <TopicCard
            key={topic.id}
            topic={topic}
            onSave={onSaveTopic}
            onDiscard={onDiscardTopic}
          />
        ))}
      </div>
    </div>
  );
}

// /hooks/use-topic-storage.ts
import { useState, useEffect } from 'react';
import { GeneratedTopic } from '@/types/topic-builder';

export function useTopicStorage() {
  const [savedTopics, setSavedTopics] = useState<GeneratedTopic[]>([]);
  
  // Load saved topics from localStorage on mount
  useEffect(() => {
    const storedTopics = localStorage.getItem('savedTopics');
    if (storedTopics) {
      try {
        setSavedTopics(JSON.parse(storedTopics));
      } catch (error) {
        console.error('Failed to parse saved topics:', error);
      }
    }
  }, []);
  
  // Save topics to localStorage whenever they change
  useEffect(() => {
    localStorage.setItem('savedTopics', JSON.stringify(savedTopics));
  }, [savedTopics]);
  
  const saveTopic = (topic: GeneratedTopic) => {
    setSavedTopics(prev => {
      // Don't add duplicates
      if (prev.some(t => t.id === topic.id)) return prev;
      return [...prev, topic];
    });
  };
  
  const removeTopic = (id: string) => {
    setSavedTopics(prev => prev.filter(topic => topic.id !== id));
  };
  
  const exportTopics = (format: 'json' | 'csv' = 'json') => {
    if (format === 'json') {
      const dataStr = JSON.stringify(savedTopics, null, 2);
      const dataUri = `data:application/json;charset=utf-8,${encodeURIComponent(dataStr)}`;
      
      const exportFileDefaultName = `topic-ideas-${new Date().toISOString().slice(0, 10)}.json`;
      
      const linkElement = document.createElement('a');
      linkElement.setAttribute('href', dataUri);
      linkElement.setAttribute('download', exportFileDefaultName);
      linkElement.click();
    } else if (format === 'csv') {
      // CSV export implementation
    }
  };
  
  return {
    savedTopics,
    saveTopic,
    removeTopic,
    exportTopics,
  };
}
