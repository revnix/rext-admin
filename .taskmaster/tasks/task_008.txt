// /lib/export-utils.ts
import { GeneratedTopic } from '@/types/topic-builder';

export function exportTopicsAsJSON(topics: GeneratedTopic[]): void {
  const dataStr = JSON.stringify(topics, null, 2);
  const dataUri = `data:application/json;charset=utf-8,${encodeURIComponent(dataStr)}`;
  
  const exportFileDefaultName = `content-topics-${new Date().toISOString().slice(0, 10)}.json`;
  
  const linkElement = document.createElement('a');
  linkElement.setAttribute('href', dataUri);
  linkElement.setAttribute('download', exportFileDefaultName);
  linkElement.click();
}

export function exportTopicsAsCSV(topics: GeneratedTopic[]): void {
  // Define CSV headers
  const headers = [
    'Title',
    'Angle',
    'Relevance Score',
    'Freshness Score',
    'Novelty Score',
    'Audience Fit',
    'Why It Works',
    'Tags'
  ];
  
  // Convert topics to CSV rows
  const rows = topics.map(topic => [
    `"${topic.title.replace(/"/g, '""')}"`,
    `"${topic.angle.replace(/"/g, '""')}"`,
    topic.scores.relevance.toFixed(2),
    topic.scores.freshness.toFixed(2),
    topic.scores.novelty.toFixed(2),
    `"${topic.audience_fit.join(', ')}"`,
    `"${topic.why_it_works.replace(/"/g, '""')}"`,
    `"${topic.tags.join(', ')}"`
  ]);
  
  // Combine headers and rows
  const csvContent = [
    headers.join(','),
    ...rows.map(row => row.join(','))
  ].join('\n');
  
  const dataUri = `data:text/csv;charset=utf-8,${encodeURIComponent(csvContent)}`;
  const exportFileDefaultName = `content-topics-${new Date().toISOString().slice(0, 10)}.csv`;
  
  const linkElement = document.createElement('a');
  linkElement.setAttribute('href', dataUri);
  linkElement.setAttribute('download', exportFileDefaultName);
  linkElement.click();
}

// /components/topic-builder/results/TopicManagement.tsx
import { useState } from 'react';
import { GeneratedTopic } from '@/types/topic-builder';
import { exportTopicsAsJSON, exportTopicsAsCSV } from '@/lib/export-utils';

interface TopicManagementProps {
  topics: GeneratedTopic[];
  savedTopics: GeneratedTopic[];
  onSaveTopic: (topic: GeneratedTopic) => void;
  onRemoveTopic: (id: string) => void;
  onRegenerateTopic: (id: string) => void;
  onEditTopic: (id: string, updates: Partial<GeneratedTopic>) => void;
}

export function TopicManagement({ 
  topics, 
  savedTopics, 
  onSaveTopic, 
  onRemoveTopic,
  onRegenerateTopic,
  onEditTopic
}: TopicManagementProps) {
  const [selectedTopics, setSelectedTopics] = useState<string[]>([]);
  const [editingTopic, setEditingTopic] = useState<string | null>(null);
  const [editForm, setEditForm] = useState<Partial<GeneratedTopic>>({});
  
  const handleSelectAll = () => {
    if (selectedTopics.length === topics.length) {
      setSelectedTopics([]);
    } else {
      setSelectedTopics(topics.map(topic => topic.id));
    }
  };
  
  const handleSelectTopic = (id: string) => {
    if (selectedTopics.includes(id)) {
      setSelectedTopics(selectedTopics.filter(topicId => topicId !== id));
    } else {
      setSelectedTopics([...selectedTopics, id]);
    }
  };
  
  const handleBulkSave = () => {
    const topicsToSave = topics.filter(topic => selectedTopics.includes(topic.id));
    topicsToSave.forEach(topic => onSaveTopic(topic));
    setSelectedTopics([]);
  };
  
  const handleBulkExport = (format: 'json' | 'csv') => {
    const topicsToExport = topics.filter(topic => selectedTopics.includes(topic.id));
    if (format === 'json') {
      exportTopicsAsJSON(topicsToExport);
    } else {
      exportTopicsAsCSV(topicsToExport);
    }
  };
  
  const startEditing = (topic: GeneratedTopic) => {
    setEditingTopic(topic.id);
    setEditForm({
      title: topic.title,
      angle: topic.angle,
      tags: [...topic.tags]
    });
  };
  
  const saveEdit = () => {
    if (editingTopic && editForm) {
      onEditTopic(editingTopic, editForm);
      setEditingTopic(null);
      setEditForm({});
    }
  };
  
  const cancelEdit = () => {
    setEditingTopic(null);
    setEditForm({});
  };
  
  return (
    <div className="topic-management">
      <div className="bulk-actions">
        <label className="select-all">
          <input
            type="checkbox"
            checked={selectedTopics.length === topics.length && topics.length > 0}
            onChange={handleSelectAll}
          />
          Select All
        </label>
        
        <div className="action-buttons">
          <button 
            onClick={handleBulkSave}
            disabled={selectedTopics.length === 0}
          >
            Save Selected
          </button>
          
          <div className="export-dropdown">
            <button 
              className="export-button"
              disabled={selectedTopics.length === 0}
            >
              Export Selected
            </button>
            <div className="export-options">
              <button onClick={() => handleBulkExport('json')}>JSON</button>
              <button onClick={() => handleBulkExport('csv')}>CSV</button>
            </div>
          </div>
        </div>
      </div>
      
      <div className="topic-list">
        {topics.map(topic => (
          <div key={topic.id} className="topic-item">
            <div className="topic-select">
              <input
                type="checkbox"
                checked={selectedTopics.includes(topic.id)}
                onChange={() => handleSelectTopic(topic.id)}
              />
            </div>
            
            {editingTopic === topic.id ? (
              <div className="topic-edit-form">
                <div className="form-field">
                  <label>Title</label>
                  <input
                    type="text"
                    value={editForm.title || ''}
                    onChange={(e) => setEditForm({ ...editForm, title: e.target.value })}
                  />
                </div>
                
                <div className="form-field">
                  <label>Angle</label>
                  <textarea
                    value={editForm.angle || ''}
                    onChange={(e) => setEditForm({ ...editForm, angle: e.target.value })}
                  />
                </div>
                
                <div className="form-field">
                  <label>Tags (comma separated)</label>
                  <input
                    type="text"
                    value={(editForm.tags || []).join(', ')}
                    onChange={(e) => {
                      const tagsString = e.target.value;
                      const tags = tagsString.split(',').map(tag => tag.trim()).filter(Boolean);
                      setEditForm({ ...editForm, tags });
                    }}
                  />
                </div>
                
                <div className="edit-actions">
                  <button onClick={saveEdit}>Save</button>
                  <button onClick={cancelEdit}>Cancel</button>
                </div>
              </div>
            ) : (
              <div className="topic-content">
                <h3>{topic.title}</h3>
                <p>{topic.angle}</p>
                
                <div className="topic-actions">
                  <button onClick={() => onSaveTopic(topic)}>
                    {savedTopics.some(t => t.id === topic.id) ? 'Saved' : 'Save'}
                  </button>
                  <button onClick={() => startEditing(topic)}>Edit</button>
                  <button onClick={() => onRegenerateTopic(topic.id)}>Regenerate</button>
                </div>
              </div>
            )}
          </div>
        ))}
      </div>
    </div>
  );
}
