// /components/ErrorBoundary.tsx
import { Component, ErrorInfo, ReactNode } from 'react';

interface ErrorBoundaryProps {
  children: ReactNode;
  fallback?: ReactNode;
}

interface ErrorBoundaryState {
  hasError: boolean;
  error?: Error;
}

export class ErrorBoundary extends Component<ErrorBoundaryProps, ErrorBoundaryState> {
  constructor(props: ErrorBoundaryProps) {
    super(props);
    this.state = { hasError: false };
  }

  static getDerivedStateFromError(error: Error): ErrorBoundaryState {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo): void {
    console.error('Error caught by boundary:', error, errorInfo);
    // Could send to error reporting service here
  }

  render(): ReactNode {
    if (this.state.hasError) {
      if (this.props.fallback) {
        return this.props.fallback;
      }
      
      return (
        <div className="error-boundary">
          <h2>Something went wrong</h2>
          <p>We're sorry, but there was an error loading this component.</p>
          <p className="error-details">{this.state.error?.message}</p>
          <button onClick={() => this.setState({ hasError: false })}>
            Try Again
          </button>
        </div>
      );
    }

    return this.props.children;
  }
}

// Enhanced version of topic generation with loading and error states
import { useState } from 'react';
import { TopicBuilderFormData, GeneratedTopic } from '@/types/topic-builder';
import { LoadingState } from '@/components/ui/LoadingState';
import { ErrorMessage } from '@/components/ui/ErrorMessage';

export function GenerationStep({ formData }: { formData: TopicBuilderFormData }) {
  const [topics, setTopics] = useState<GeneratedTopic[]>([]);
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [retryCount, setRetryCount] = useState(0);
  
  const generateTopics = async () => {
    setIsLoading(true);
    setError(null);
    
    try {
      const response = await fetch('/api/generate-topics', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          formData,
          timestamp: new Date().toISOString(),
          requestId: `req-${Date.now()}-${Math.random().toString(36).substring(2, 9)}`,
        }),
      });
      
      if (!response.ok) {
        const errorData = await response.json();
        throw new Error(errorData.error || 'Failed to generate topics');
      }
      
      const data = await response.json();
      setTopics(data.topics);
    } catch (err) {
      console.error('Topic generation error:', err);
      setError(err instanceof Error ? err.message : 'An unexpected error occurred');
    } finally {
      setIsLoading(false);
    }
  };
  
  const handleRetry = () => {
    setRetryCount(prev => prev + 1);
    generateTopics();
  };
  
  // Auto-generate on first render
  useEffect(() => {
    generateTopics();
  }, []);
  
  if (isLoading) {
    return (
      <div className="generation-step">
        <LoadingState 
          message={`Generating ${formData.num_ideas} content topics for you...`} 
          size="large" 
        />
        <div className="generation-info">
          <p>This may take up to 15 seconds.</p>
          <p>We're analyzing your inputs and creating tailored content topics.</p>
        </div>
      </div>
    );
  }
  
  if (error) {
    return (
      <div className="generation-step error">
        <ErrorMessage 
          error={`Failed to generate topics: ${error}`} 
          onRetry={handleRetry} 
        />
        <div className="error-help">
          <p>This could be due to:</p>
          <ul>
            <li>Temporary service disruption</li>
            <li>Network connectivity issues</li>
            <li>High server load</li>
          </ul>
          <p>You've tried {retryCount} time(s). Please try again or come back later.</p>
        </div>
      </div>
    );
  }
  
  return (
    <div className="generation-step success">
      <div className="generation-header">
        <h2>Generated Topics</h2>
        <button onClick={generateTopics} className="regenerate-button">
          Regenerate All
        </button>
      </div>
      
      {topics.length > 0 ? (
        <TopicList 
          topics={topics} 
          onSaveTopic={(id) => console.log('Save topic:', id)} 
          onDiscardTopic={(id) => console.log('Discard topic:', id)} 
        />
      ) : (
        <div className="no-topics">
          <p>No topics were generated. Please try again with different inputs.</p>
          <button onClick={generateTopics}>Generate Topics</button>
        </div>
      )}
    </div>
  );
}
